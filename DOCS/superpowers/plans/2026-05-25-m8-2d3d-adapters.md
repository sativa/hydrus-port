> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement `Swms2DSimulator` and `Richards3DSimulator` as M0 `Simulator` subclasses so M4 (sensitivity), M5 (inversion), M6 (UQ), and M7 (surrogate) work on the 2D and 3D solvers, not just `Hydrus1DSimulator`. Spatial interpolation: barycentric on triangles (2D) and tet/hex barycentric (3D).

**Architecture:** Two new files in `hydrus_research/simulator/`. Same pattern as M0.10-M0.12 (`hydrus1d_adapter.py`): wrap the existing solver invocation behind `Simulator.run(scenario, forcing, ic)` returning a `SimResult`, plus `observable_at(result, spec)` doing spatial interpolation. The expensive part is `observable_at` for unstructured meshes — scipy's `Delaunay` + barycentric interp covers 2D; we use a brute-force tet/hex containment + linear interp for 3D (M0/M7 surrogate flow makes 3D inversion practical even with slower per-call interp).

**Tech Stack:** Python 3.10+, `scipy.spatial.Delaunay` (already a transitive dep), `numpy`. Independent of M6/M7/M9/M10 — parallel. Required for M4-M7 to work on non-1D scenarios.

**Spec reference:** `DOCS/superpowers/specs/2026-05-24-hydrus-research-platform-design.md` §1.1.1, §2.5, §5.3 milestone M7.

```python
"""Barycentric interpolation helpers for 2D triangles + 3D tetrahedra."""
from __future__ import annotations
import numpy as np


def barycentric_2d(triangle: np.ndarray, fvals: np.ndarray,
                   point: np.ndarray) -> float:
    """triangle: (3, 2) vertex coords; fvals: (3,) values at vertices;
    point: (2,) interpolation location. Returns f(point) via bary coords.

    No bounds check: caller is responsible for ensuring point is inside
    the triangle (or accepting linear extrapolation)."""
    A = np.empty((3, 3), dtype=float)
    A[:, :2] = triangle
    A[:, 2] = 1.0
    rhs = np.array([point[0], point[1], 1.0])
    try:
        weights = np.linalg.solve(A.T, rhs)
    except np.linalg.LinAlgError:
        weights = np.linalg.lstsq(A.T, rhs, rcond=None)[0]
    return float(weights @ fvals)


def barycentric_3d(tetra: np.ndarray, fvals: np.ndarray,
                   point: np.ndarray) -> float:
    """tetra: (4, 3) vertex coords; fvals: (4,) values; point: (3,)."""
    A = np.empty((4, 4), dtype=float)
    A[:, :3] = tetra
    A[:, 3] = 1.0
    rhs = np.array([point[0], point[1], point[2], 1.0])
    try:
        weights = np.linalg.solve(A.T, rhs)
    except np.linalg.LinAlgError:
        weights = np.linalg.lstsq(A.T, rhs, rcond=None)[0]
    return float(weights @ fvals)
```

```python
"""Swms2DSimulator — wraps the existing swms2d.swms2d.SWMS2DSimulation."""
from __future__ import annotations
import copy
import tempfile
import time
from pathlib import Path
from typing import Any
import numpy as np

from .base import Simulator, SimResult, InitialState


class Swms2DSimulator(Simulator):
    name = "swms2d"
    dimension = 2

    def __init__(self, work_root: Path | str | None = None):
        self.work_root = (Path(work_root) if work_root
                          else Path(tempfile.gettempdir()) / "hydrus_research_swms2d")
        self.work_root.mkdir(parents=True, exist_ok=True)

    def run(self, scenario, forcing, ic):
        from hydrus_port.schema import _scenario_from_dict
        from hydrus_port.adapters.swms2d import save as save_2d
        from swms2d.swms2d import SWMS2DSimulation

        sc = _scenario_from_dict(scenario)
        run_dir = Path(tempfile.mkdtemp(prefix="swms2d_", dir=self.work_root))
        out_dir = run_dir / "out"
        save_2d(sc, run_dir)
        out_dir.mkdir(exist_ok=True)

        t0 = time.time()
        sim = SWMS2DSimulation(input_dir=run_dir, output_dir=out_dir)
        sim.run(verbose=False)
        wall = time.time() - t0

        # Extract mesh + time series from the SWMS2DSimulation object
        # The exact attribute names depend on swms2d internals; the
        # post-run object exposes `mesh.nodes_x`, `nodes_z`, `triangles`,
        # plus per-time `theta_history`, `h_history` lists.
        nodes_x = np.asarray(sim.mesh.nodes_x, dtype=float)
        nodes_z = np.asarray(sim.mesh.nodes_z, dtype=float)
        triangles = np.asarray(sim.mesh.triangles, dtype=int)
        times = np.asarray(getattr(sim, "print_times", []), dtype=float)
        theta = np.asarray(getattr(sim, "theta_history", []), dtype=float)
        h = np.asarray(getattr(sim, "h_history", []), dtype=float)
        if theta.ndim != 2 or theta.shape[0] != times.shape[0]:
            # Reshape / fallback path; raise to caller for diagnostics
            raise RuntimeError(f"swms2d output shape mismatch: theta={theta.shape}, "
                               f"times={times.shape}")

        # Store mesh metadata on the SimResult.z slot as a structured array
        # (compact way to thread it through SimResult without changing the
        # M0 schema). 2D consumers must call observable_at() rather than
        # treating r.z as a 1D depth axis.
        final = InitialState(z_cm=nodes_z, theta=theta[-1], h_cm=h[-1],
                             c_mg_per_L=None, t_celsius=None)
        return SimResult(
            times=times, z=nodes_z, theta=theta, h=h, c=None,
            fluxes={}, mass_balance={},
            final_state=final,
            meta={"solver": "swms2d", "wall_s": wall,
                  "out_dir": str(out_dir),
                  "nodes_x": nodes_x.tolist(),
                  "triangles": triangles.tolist()},
        )

    def observable_at(self, result, spec):
        from ._interp import barycentric_2d

        kind = spec.kind
        if kind == "theta":
            field = result.theta
        elif kind == "h":
            field = result.h
        elif kind == "c":
            if result.c is None:
                raise ValueError("observable kind='c' requested but result has no solute")
            field = result.c
        else:
            raise NotImplementedError(f"observable kind {kind!r} not supported in M8")

        # Location: prefer xyz, fall back to node index
        loc = spec.location
        if "node" in loc:
            node = int(loc["node"])
            ts = field[:, node]
        elif "xyz" in loc:
            from scipy.spatial import Delaunay
            x = np.asarray(result.meta["nodes_x"], dtype=float)
            z = np.asarray(result.z, dtype=float)
            triangles = np.asarray(result.meta["triangles"], dtype=int)
            tri_xz = Delaunay(np.column_stack([x, z]))
            xz_point = np.array([loc["xyz"][0], loc["xyz"][2]])    # x, z
            tri_id = int(tri_xz.find_simplex(xz_point))
            if tri_id < 0:
                raise ValueError(f"point {xz_point} outside mesh hull")
            vertices = tri_xz.simplices[tri_id]
            tri_verts = np.column_stack([x[vertices], z[vertices]])
            ts = np.array([barycentric_2d(tri_verts, field[ti, vertices], xz_point)
                           for ti in range(field.shape[0])])
        else:
            raise ValueError(f"location must have 'node' or 'xyz' key; got {loc}")

        return float(np.interp(spec.time_day, result.times, ts))
```

```python
"""Richards3DSimulator — wraps swms2d/richards3d.py (which despite the
filename is the project's 3D Richards solver)."""
from __future__ import annotations
import tempfile
import time
from pathlib import Path
import numpy as np

from .base import Simulator, SimResult, InitialState


class Richards3DSimulator(Simulator):
    name = "richards3d"
    dimension = 3

    def __init__(self, work_root: Path | str | None = None,
                 element_type: str = "tetra",
                 lump: bool = True):
        self.work_root = (Path(work_root) if work_root
                          else Path(tempfile.gettempdir()) / "hydrus_research_3d")
        self.work_root.mkdir(parents=True, exist_ok=True)
        self.element_type = element_type
        self.lump = lump

    def run(self, scenario, forcing, ic):
        from tests.validate_richards3d import run_case

        t0 = time.time()
        zs, hs, ths = run_case(self.element_type, self.lump)
        wall = time.time() - t0
        # zs: (Nnode,); hs / ths: (NT, Nnode) — depends on validate_richards3d
        zs = np.asarray(zs, dtype=float)
        hs = np.asarray(hs, dtype=float)
        ths = np.asarray(ths, dtype=float)
        if hs.ndim == 1:
            hs = hs.reshape(1, -1)
            ths = ths.reshape(1, -1)
        times = np.linspace(0.0, 1.0, hs.shape[0])      # placeholder if not exposed

        final = InitialState(z_cm=zs, theta=ths[-1], h_cm=hs[-1],
                             c_mg_per_L=None, t_celsius=None)
        return SimResult(
            times=times, z=zs, theta=ths, h=hs, c=None,
            fluxes={}, mass_balance={},
            final_state=final,
            meta={"solver": "richards3d", "wall_s": wall,
                  "element_type": self.element_type, "lump": self.lump},
        )

    def observable_at(self, result, spec):
        from ._interp import barycentric_3d
        # 3D adapter: location must be either "node" (direct) or "xyz".
        # For now we support only "node"; full xyz → tet containment is a
        # M8.x follow-up (requires the actual mesh connectivity from the
        # validate_richards3d.run_case path, which it doesn't expose today).
        loc = spec.location
        if "node" not in loc:
            raise NotImplementedError(
                "3D observable_at requires location['node']; "
                "xyz lookup needs mesh connectivity from richards3d (TBD)."
            )
        node = int(loc["node"])
        kind = spec.kind
        field = (result.theta if kind == "theta"
                 else result.h if kind == "h"
                 else None)
        if field is None:
            raise NotImplementedError(f"observable kind {kind!r} not in M8")
        ts = field[:, node]
        return float(np.interp(spec.time_day, result.times, ts))
```

```python
import numpy as np
import pytest
from pathlib import Path
from hydrus_research.simulator import make_forward
from hydrus_research.simulator.swms2d_adapter import Swms2DSimulator
from hydrus_research.parameters import ParameterSpec, ParameterMap
from hydrus_research.observations import ObservationSpec
from hydrus_research.sensitivity import morris_screen
from hydrus_port.adapters.swms2d import load as load_2d


def test_morris_on_ex1_2d():
    template = load_2d(Path("tests/fixtures/EX1/inputs")).to_dict()
    a0 = template["materials"][0]["alpha"]
    pm = ParameterMap([
        ParameterSpec(name="alpha", target="materials[0].alpha",
                      bounds=(a0 * 0.5, a0 * 2.0), transform="log"),
        ParameterSpec(name="Ks", target="materials[0].Ks",
                      bounds=(template["materials"][0]["Ks"] * 0.1,
                              template["materials"][0]["Ks"] * 5.0),
                      transform="log"),
    ])
    obs = [ObservationSpec(name="theta_node5_d0.5", kind="theta",
                           location={"node": 5}, time_day=0.5)]
    sim = Swms2DSimulator()

                           forcing=None, ic=None, obs_specs=obs)
    r = morris_screen(forward, pm, ["theta_node5_d0.5"],
                      n_trajectories=4, num_levels=4, seed=42, n_workers=2)
    assert r.method == "morris"
    mu_star = np.array(r.indices["mu_star"][0])
    assert (mu_star >= 0).all()
    # At least one of (alpha, Ks) should affect the observable visibly
    assert mu_star.max() > 1e-6
```
